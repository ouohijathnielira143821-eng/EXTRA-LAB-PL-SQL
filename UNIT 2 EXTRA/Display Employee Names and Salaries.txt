DECLARE
    CURSOR c1 IS SELECT ename, sal FROM emp;
    v_name emp.ename%TYPE;
    v_sal emp.sal%TYPE;
BEGIN
    OPEN c1;
    LOOP
        FETCH c1 INTO v_name, v_sal;
        EXIT WHEN c1%NOTFOUND;
        DBMS_OUTPUT.PUT_LINE(v_name || ' ' || v_sal);
    END LOOP;
    CLOSE c1;
END;
/
