DECLARE
    v_id emp.empno%TYPE := &emp_id;
    v_name emp.ename%TYPE;
BEGIN
    SELECT ename INTO v_name
    FROM emp WHERE empno = v_id;

    DBMS_OUTPUT.PUT_LINE('Employee: ' || v_name);
EXCEPTION
    WHEN NO_DATA_FOUND THEN
        DBMS_OUTPUT.PUT_LINE('Employee does not exist');
END;
/
