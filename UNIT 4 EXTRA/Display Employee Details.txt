CREATE OR REPLACE PROCEDURE emp_details(
    p_id IN NUMBER
)
IS
    v_name emp.ename%TYPE;
    v_sal emp.sal%TYPE;
BEGIN
    SELECT ename, sal INTO v_name, v_sal
    FROM emp WHERE empno = p_id;

    DBMS_OUTPUT.PUT_LINE('Name: ' || v_name);
    DBMS_OUTPUT.PUT_LINE('Salary: ' || v_sal);
END;
/
